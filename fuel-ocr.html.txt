<script>
const FuelOCR = {
  _listener: null,
  _currentImageBlob: null,
  _currentImageUrl: '',

  init() {
    document.getElementById('fuelDate').value = U.today();
    document.getElementById('fTime').value = U.nowTime();
    document.getElementById('receiptFile').addEventListener('change', e => this.onFile(e));
    document.getElementById('fuelDate').addEventListener('change', () => this.switchDate());
    this.switchDate();
  },

  onFile(e) {
    const file = e.target.files[0];
    if (!file) return;
    this._currentImageBlob = file;
    const url = URL.createObjectURL(file);
    this._currentImageUrl = url;
    const img = document.getElementById('preview');
    img.src = url;
    img.classList.add('show');
    document.getElementById('btnOcr').disabled = false;
    document.getElementById('ocrRaw').classList.remove('show');
    U.toast('✅ 已載入圖片，可開始 OCR', 'ok');
  },

  async run() {
    if (!this._currentImageBlob) { U.toast('請先選擇圖片', 'warn'); return; }
    if (typeof Tesseract === 'undefined') { U.toast('OCR 庫未載入', 'err'); return; }
    const btn = document.getElementById('btnOcr');
    btn.disabled = true;
    U.loading(true, 'OCR 辨識中，請稍等...');
    try {
      const { data: { text } } = await Tesseract.recognize(
        this._currentImageBlob, 'chi_tra+eng',
        { logger: m => {
          if (m.status === 'recognizing text') {
            document.getElementById('loadingText').textContent =
              `OCR 辨識中... ${Math.round((m.progress||0)*100)}%`;
          }
        }}
      );
      document.getElementById('ocrRaw').textContent = text || '（無辨識結果）';
      document.getElementById('ocrRaw').classList.add('show');
      this.fillFromText(text);
      U.toast('✅ OCR 完成，請確認資料', 'ok');
    } catch (e) {
      console.error(e);
      U.toast('OCR 失敗：' + e.message, 'err');
    } finally {
      U.loading(false);
      btn.disabled = false;
    }
  },

  fillFromText(text) {
    if (!text) return;
    const t = text.replace(/\s+/g, ' ');
    const amountPatterns = [
      /(?:總額|合計|金額|應付|TOTAL|Total|AMOUNT)[^\d]{0,10}(\d{1,4}[.,]\d{2})/,
      /\$\s*(\d{1,4}[.,]\d{2})/,
      /(\d{1,4}\.\d{2})/
    ];
    for (const p of amountPatterns) {
      const m = t.match(p);
      if (m) { document.getElementById('fAmount').value = parseFloat(m[1].replace(',', '.')); break; }
    }
    const plateMatch = t.match(/\b([A-Z]{1,2})\s?(\d{1,4})\b/);
    if (plateMatch) document.getElementById('fPlate').value = plateMatch[1] + plateMatch[2];
    const timeMatch = t.match(/\b([0-2]?\d):([0-5]\d)\b/);
    if (timeMatch) document.getElementById('fTime').value = U.pad(timeMatch[1]) + ':' + timeMatch[2];
    const stations = ['中石化', 'Esso', 'Shell', '加德士', '蜆殼', '埃索'];
    for (const s of stations) {
      if (t.includes(s)) {
        const sel = document.getElementById('fStation');
        for (const opt of sel.options) {
          if (opt.value === s || opt.text === s) { sel.value = opt.value; break; }
        }
        break;
      }
    }
    if (t.includes('石油氣') || t.includes('LPG')) document.getElementById('fFuelType').value = '石油氣';
    else if (t.includes('柴油') || t.includes('Diesel')) document.getElementById('fFuelType').value = '柴油';
    else if (t.includes('汽油') || t.includes('Petrol')) document.getElementById('fFuelType').value = '汽油';
  },

  // ====== 新增：圖片壓縮工具 ======
  compressImage(file, maxWidth = 800, quality = 0.7) {
    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.readAsDataURL(file);
      reader.onload = (e) => {
        const img = new Image();
        img.src = e.target.result;
        img.onload = () => {
          const canvas = document.createElement('canvas');
          let width = img.width;
          let height = img.height;
          if (width > maxWidth) {
            height = Math.round(height * (maxWidth / width));
            width = maxWidth;
          }
          canvas.width = width;
          canvas.height = height;
          const ctx = canvas.getContext('2d');
          ctx.drawImage(img, 0, 0, width, height);
          // 轉為 JPEG 格式並壓縮
          const dataUrl = canvas.toDataURL('image/jpeg', quality);
          resolve(dataUrl);
        };
        img.onerror = reject;
      };
      reader.onerror = reject;
    });
  },

  async save() {
    const date = document.getElementById('fuelDate').value;
    const amount = parseFloat(document.getElementById('fAmount').value);
    if (!date) { U.toast('請選擇日期', 'warn'); return; }
    if (!amount || amount <= 0) { U.toast('請輸入有效金額', 'warn'); return; }

    const fuel = {
      station: document.getElementById('fStation').value,
      plate: document.getElementById('fPlate').value.trim(),
      amount: amount,
      fuelType: document.getElementById('fFuelType').value,
      time: document.getElementById('fTime').value,
      remark: document.getElementById('fRemark').value.trim(),
      source: 'ocr'
    };

    U.loading(true, '儲存中...');
    try {
      // ====== 替代方案：將圖片轉為 Base64 並存入 Realtime Database ======
      if (this._currentImageBlob) {
        try {
          U.loading(true, '壓縮圖片並轉換中...');
          const base64String = await this.compressImage(this._currentImageBlob, 800, 0.7);
          fuel.receiptBase64 = base64String;
        } catch (e) {
          console.warn('圖片壓縮失敗，繼續儲存文字資料：', e);
        }
      }

      await DataLayer.addFuel(date, fuel);
      U.toast('✅ 燃料費已儲存（含圖片）', 'ok');
      this.clear();
    } catch (e) {
      U.toast('儲存失敗：' + e.message, 'err');
    } finally {
      U.loading(false);
    }
  },

  clear() {
    document.getElementById('receiptFile').value = '';
    document.getElementById('preview').classList.remove('show');
    document.getElementById('ocrRaw').classList.remove('show');
    document.getElementById('fStation').value = '';
    document.getElementById('fPlate').value = '';
    document.getElementById('fAmount').value = '';
    document.getElementById('fFuelType').value = '石油氣';
    document.getElementById('fTime').value = U.nowTime();
    document.getElementById('fRemark').value = '';
    document.getElementById('btnOcr').disabled = true;
    this._currentImageBlob = null;
  },

  switchDate() {
    const date = document.getElementById('fuelDate').value;
    if (!date) return;
    if (this._listener) this._listener();
    this._listener = DataLayer.watchDayFuel(date, data => this.render(data));
  },

  render(data) {
    const tb = document.getElementById('fuelBody');
    const keys = Object.keys(data || {});
    if (!keys.length) {
      tb.innerHTML = '<tr><td colspan="7">暫無紀錄</td></tr>';
      document.getElementById('fuelDaySummary').textContent = '當日燃料費：HK$0.00';
      return;
    }
    let total = 0;
    tb.innerHTML = keys.map(k => {
      const f = data[k];
      total += f.amount || 0;
      return `<tr>
        <td>${U.esc(f.time || '-')}</td>
        <td>${U.esc(f.station || '-')}</td>
        <td>${U.esc(f.plate || '-')}</td>
        <td>${U.esc(f.fuelType || '-')}</td>
        <td>$${U.fmt(f.amount)}</td>
        <td>${U.esc(f.remark || '')}</td>
        <td>
          ${f.receiptBase64 ? `<a href="${f.receiptBase64}" target="_blank" class="btn btn-ghost btn-sm" style="text-decoration:none;">📷 查看</a>` : ''}
          <button class="btn btn-err btn-sm" onclick="FuelOCR.del('${U.esc(k)}')">🗑</button>
        </td>
      </tr>`;
    }).join('');
    document.getElementById('fuelDaySummary').textContent = `當日燃料費：HK$${U.fmt(total)}`;
  },

  async del(id) {
    if (!U.confirm('確定刪除此燃料費紀錄？')) return;
    const date = document.getElementById('fuelDate').value;
    try {
      await DataLayer.deleteFuel(date, id);
      U.toast('已刪除', 'ok');
    } catch (e) {
      U.toast('刪除失敗：' + e.message, 'err');
    }
  }
};

FuelOCR.init();
</script>

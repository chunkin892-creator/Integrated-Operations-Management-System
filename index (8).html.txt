function getDateRange(mode){
  const t=new Date(),y=t.getFullYear(),m=t.getMonth(),ms=String(m+1).padStart(2,'0');
  if(mode==='firstHalf')return{start:`${y}-${ms}-01`,end:`${y}-${ms}-15`,label:`${y}年${m+1}月 上半個月 (1–15號)`};
  if(mode==='secondHalf'){const ld=new Date(y,m+1,0).getDate();return{start:`${y}-${ms}-16`,end:`${y}-${ms}-${String(ld).padStart(2,'0')}`,label:`${y}年${m+1}月 下半個月 (16–${ld}號)`};}
  const s=document.getElementById('luxStartDate').value,e=document.getElementById('luxEndDate').value;
  if(!s||!e){alert('請選擇自定開始同結束日期');return null;}
  if(s>e){alert('開始日期唔可以遲過結束日期');return null;}
  return{start:s,end:e,label:`自定範圍：${s} 至 ${e}`};
}

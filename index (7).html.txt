function generateDefaultRoster(){const mv=document.getElementById('selectMonth').value;if(!mv)return;const dM=document.getElementById('driverMorning').value.trim()||'早更司機';const dN=document.getElementById('driverNight').value.trim()||'夜更司機';const dS=document.getElementById('driverSpecial').value.trim()||'特更司機';const ss=document.getElementById('firstSatDriver');let sdt=ss?ss.value:dM;if(!sdt)sdt=dM;const[y,m]=mv.split('-').map(Number);const td=new Date(y,m,0).getDate();const wh=['星期日','星期一','星期二','星期三','星期四','星期五','星期六'];currentCalculatedData=[];for(let d=1;d<=td;d++){const ds=`${y}-${String(m).padStart(2,'0')}-${String(d).padStart(2,'0')}`;const dobj=new Date(y,m-1,d);const wd=dobj.getDay();const hn=HK_HOLIDAYS[ds];const ih=!!hn||wd===0;let rd={date:ds,dayNum:d,weekDayStr:wh[wd],holidayName:hn||(wd===0?'週日':''),isHoliday:ih,mDriver:'',mTime:'',mHours:0,nDriver:'',nTime:'',nHours:0,dailyHours:0,dailyWage:0};if(ih){rd.nDriver=dS;rd.nTime='08:00 - 19:00 (11小時)';rd.nHours=11;rd.dailyHours=11;}else if(wd===6){rd.mDriver=dS;rd.mTime='07:00 - 13:00 (6小時)';rd.mHours=6;rd.nDriver=sdt;rd.nTime='13:00 - 19:00 (6小時)';rd.nHours=6;rd.dailyHours=12;sdt=(sdt===dM)?dN:dM;}else{rd.mDriver=dM;rd.mTime='06:00 - 14:00 (8小時)';rd.mHours=8;rd.nDriver=dN;rd.nTime='14:00 - 21:00 (7小時)';rd.nHours=7;rd.dailyHours=15;}currentCalculatedData.push(rd);}saveAndRender();}
function saveAndRender(){const mv=document.getElementById('selectMonth').value;if(!mv)return;safeSet(`minibus_roster_${mv}`,JSON.stringify(currentCalculatedData));updateStatus('🟡 雲端儲存中...','syncing');db.ref(`rosters/${mv}`).set(currentCalculatedData).then(()=>updateStatus('🟢 雲端已連線 (實時同步)','online')).catch(err=>{console.error(err);updateStatus('🔴 雲端儲存失敗 (保留本地)','offline');});recalculateAll();}
function recalculateAll(){const rate=parseFloat(document.getElementById('hourlyRate').value)||75;currentStats={};currentCalculatedData.forEach(row=>{row.dailyHours=(parseFloat(row.mHours)||0)+(parseFloat(row.nHours)||0);row.dailyWage=row.dailyHours*rate;const fh=row.dayNum<=15;[{name:row.mDriver,h:parseFloat(row.mHours)||0},{name:row.nDriver,h:parseFloat(row.nHours)||0}].forEach(sh=>{if(sh.name&&sh.name.trim()!==''&&sh.name!=='-'){const sn=sh.name.trim();if(!currentStats[sn])currentStats[sn]={h1:0,h2:0,totalH:0};if(fh)currentStats[sn].h1+=sh.h;else currentStats[sn].h2+=sh.h;currentStats[sn].totalH+=sh.h;}});});renderCalendar();renderSummary(currentStats,rate);}

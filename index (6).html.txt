function checkOutFromTable(key){

  if(record.checkOut){showToast('⚠️ 已經簽退！','warning');return;}

  isProcessing=true;
  const rate=getHourlyRate();const now=new Date();
  const timeStr=String(now.getHours()).padStart(2,'0')+':'+String(now.getMinutes()).padStart(2,'0');
  const inParts=(record.checkIn||'00:00').split(':');
  const inMin=parseInt(inParts[0])*60+parseInt(inParts[1]);

  let diff=outMin-inMin;if(diff<0)diff+=1440;
